cmake_minimum_required(VERSION 3.16)
project(ntru_oqs_bridge C ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_BUILD_RPATH "${OQS_BUILD}/lib")
set(CMAKE_INSTALL_RPATH "$ORIGIN")
if(NR_IMPL STREQUAL "avx2")
  set(PQDIR "${OQS_SOURCE}/src/kem/ntru/pqclean_ntruhrss701_avx2")
  set(MULFILE "${PQDIR}/poly_rq_mul.s")
  set(MULFN PQCLEAN_NTRUHRSS701_AVX2_poly_Rq_mul)
  add_compile_options(-mavx2 -mbmi2)
else()
  set(PQDIR "${OQS_SOURCE}/src/kem/ntru/pqclean_ntruhrss701_clean")
  set(MULFILE "${PQDIR}/poly_rq_mul.c")
  set(MULFN PQCLEAN_NTRUHRSS701_CLEAN_poly_Rq_mul)
endif()
add_library(ntru_rq_bridge SHARED rq_bridge.c ${MULFILE})
target_include_directories(ntru_rq_bridge PRIVATE "${PQDIR}")
target_compile_definitions(ntru_rq_bridge PRIVATE NR_IMPLEMENTATION="${NR_IMPL}" NR_RQ_MUL=${MULFN})
add_library(ntru_kem_bridge SHARED kem_bridge.c)
target_include_directories(ntru_kem_bridge PRIVATE "${OQS_BUILD}/include")
target_link_libraries(ntru_kem_bridge PRIVATE "${OQS_BUILD}/lib/liboqs.so" dl)
target_compile_definitions(ntru_kem_bridge PRIVATE NR_IMPLEMENTATION="${NR_IMPL}")
add_executable(bridge_smoke smoke.c)
target_link_libraries(bridge_smoke PRIVATE ntru_kem_bridge ntru_rq_bridge)
install(TARGETS ntru_kem_bridge ntru_rq_bridge LIBRARY DESTINATION lib)
install(TARGETS bridge_smoke RUNTIME DESTINATION bin)
